cmake_minimum_required(VERSION 3.21)
project(LookAtLANng VERSION 1.3.0 LANGUAGES C)
set(RELEASE_SUFFIX "")
string(TIMESTAMP BUILD_YEAR "%Y")
configure_file(src/resources/build_info.h.in generated/build_info.h @ONLY)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(scanner_core src/core/network.c src/core/host.c src/core/scanner.c src/core/ports.c src/core/vendor.c src/core/monitor.c src/core/profile.c src/core/report.c src/core/statistics.c src/core/details.c src/core/nbns.c)
target_include_directories(scanner_core PUBLIC src/core)
if(MSVC)
  add_compile_options(/W4 /utf-8)
  target_compile_options(scanner_core PRIVATE /W4 /utf-8 $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
  target_compile_definitions(scanner_core PRIVATE _CRT_SECURE_NO_WARNINGS)
  add_compile_options($<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
  add_link_options(/NXCOMPAT /DYNAMICBASE $<$<CONFIG:Release>:/LTCG>)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    add_link_options(/HIGHENTROPYVA)
  endif()
endif()
if(WIN32)
  enable_language(RC)
  set_source_files_properties(src/resources/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/generated/build_info.h;${CMAKE_CURRENT_SOURCE_DIR}/src/resources/app.manifest;${CMAKE_CURRENT_SOURCE_DIR}/data/oui.bin;${CMAKE_CURRENT_SOURCE_DIR}/src/resources/assets/app.ico")
  add_executable(LookAtLANng WIN32 src/platform/windows/win_main.c src/platform/windows/win_toolbar_icons.c src/platform/windows/win_net.c src/platform/windows/win_neighbor.c src/platform/windows/win_ping.c src/platform/windows/win_resolver.c src/platform/windows/win_nbns.c src/platform/windows/win_tools.c src/platform/windows/win_files.c src/platform/windows/win_profile_ui.c src/resources/app.rc)
  set_target_properties(LookAtLANng PROPERTIES OUTPUT_NAME "Look@LAN-ng")
  target_compile_definitions(LookAtLANng PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_include_directories(LookAtLANng PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  if(MSVC)
    target_link_options(LookAtLANng PRIVATE /MANIFEST:NO)
  endif()
  target_link_libraries(LookAtLANng PRIVATE scanner_core comctl32 ws2_32 iphlpapi dnsapi user32 gdi32 shell32 comdlg32)
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  if(NOT DESKTOP_HOST_PYTHON)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(DESKTOP_HOST_PYTHON "${Python3_EXECUTABLE}")
  endif()
  execute_process(COMMAND "${DESKTOP_HOST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed-vendor.py" "${CMAKE_CURRENT_SOURCE_DIR}/data/oui.bin" "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop_vendor.c" COMMAND_ERROR_IS_FATAL ANY)
  set(PORT_RELEASE_SUFFIX "${RELEASE_SUFFIX}")
  set(RELEASE_SUFFIX "-rc.2")
  set(PORT_VERSION "${PROJECT_VERSION}${RELEASE_SUFFIX}")
  configure_file(src/resources/build_info.h.in generated/desktop/build_info.h @ONLY)
  set(RELEASE_SUFFIX "${PORT_RELEASE_SUFFIX}")
  add_library(desktop_backend src/platform/posix/posix_net.c src/platform/posix/posix_neighbors.c src/platform/posix/posix_names.c src/platform/posix/posix_tools.c src/platform/posix/desktop_io.c src/platform/posix/desktop_app.c "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop_vendor.c")
  target_include_directories(desktop_backend PUBLIC src/platform/posix "${CMAKE_CURRENT_BINARY_DIR}/generated/desktop")
  target_link_libraries(desktop_backend PUBLIC scanner_core Threads::Threads)
  target_compile_definitions(desktop_backend PUBLIC _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE _DARWIN_C_SOURCE)
  target_compile_options(desktop_backend PRIVATE -Wall -Wextra -Wpedantic)
  if(APPLE)
    target_link_libraries(desktop_backend PUBLIC "-framework SystemConfiguration" "-framework CoreFoundation")
    enable_language(OBJC)
    add_executable(LookAtLANng MACOSX_BUNDLE src/platform/macos/main.m)
    target_compile_options(LookAtLANng PRIVATE -fobjc-arc -Wall -Wextra -Wno-deprecated-declarations)
    target_link_libraries(LookAtLANng PRIVATE desktop_backend "-framework Cocoa")
    set_target_properties(LookAtLANng PROPERTIES OUTPUT_NAME "Look@LAN-ng" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/resources/macos/Info.plist.in")
  else()
    if(LOOKLAN_TARGET_GTK)
      include("${LOOKLAN_TARGET_GTK}")
    else()
      find_package(PkgConfig REQUIRED)
      pkg_check_modules(GTK3 REQUIRED gtk+-3.0)
    endif()
    add_executable(LookAtLANng src/platform/linux/main.c)
    target_include_directories(LookAtLANng PRIVATE ${GTK3_INCLUDE_DIRS})
    target_link_directories(LookAtLANng PRIVATE ${GTK3_LIBRARY_DIRS})
    target_link_libraries(LookAtLANng PRIVATE desktop_backend ${GTK3_LIBRARIES})
    set_target_properties(scanner_core desktop_backend LookAtLANng PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_link_options(LookAtLANng PRIVATE -pie -Wl,-z,relro,-z,now,-z,noexecstack)
    set_target_properties(LookAtLANng PROPERTIES OUTPUT_NAME "Look@LAN-ng")
  endif()
endif()
